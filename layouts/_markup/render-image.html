{{- $imgSrc := .Destination -}} {{- /* MarkdownのURL（例: /images/xxx.jpg）の先頭の "/" を取り除いて assets フォルダ内を検索 */ -}} {{- $image := resources.Get (strings.TrimPrefix "/" $imgSrc) -}} {{- if $image -}} {{- $resized := $image -}} {{- /* Hugoで処理できる画像だけ幅を調べる。SVGなどは元のリソースを使う。 */ -}} {{- $processableFormats := slice "bmp" "gif" "jpeg" "png" "tiff" "webp" -}} {{- if and (in $processableFormats $image.MediaType.SubType) (gt $image.Width 800) -}} {{- $resized = $image.Resize "800x" -}} {{- end -}} {{- /* リサイズした（またはそのままの）画像を出力 */ -}} {{- else -}} {{- /* assets内に見つからない場合（外部リンク等）は元のURLを使う。 */ -}} {{- end -}}
